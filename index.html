    Web words
    Front-endThe part of a website you see and click: HTML, CSS and JavaScript running in the browser.
    Back-end / APIA program on a server that keeps the data (products, users, orders). The front-end asks it for data and sends it changes.
    EndpointOne address of the API, for example GET /api/products (“give me the products”).
    GET, POST, PUT, DELETEWhat you ask the API to do: read, add, change, remove.
    JSONA text format for data, like { "name": "Bangus", "price": 190 }. The API and the page send each other JSON.
    fetch()The JavaScript command that sends a request to an API and waits for the answer.
    Status codeThe number the API answers with: 200 = OK, 201 = created, 401 = not logged in, 403 = not allowed, 404 = not found, 422 = the form has mistakes.
    TokenA long secret code the API gives you when you log in. You send it with every request that needs a login, in the Authorization: Bearer … header.
    RoleThe kind of user: customer or seller. Each role may use different pages.
    DOMThe page as JavaScript sees it. “DOM manipulation” means JavaScript adding, changing, hiding or removing things on the page.
    EventSomething that happens on the page, like a click, typing, or a form being sent. addEventListener tells JavaScript what to do when it happens.
    ResponsiveThe layout changes to fit phones, tablets and computers.
    BootstrapA ready-made set of CSS classes and components (grid, cards, buttons, modals) so you don't write all the CSS yourself.
    ModalA pop-up window on top of the page, for example the product details or the “Add product” form.
    Vue.jsA JavaScript framework: you describe what the screen should show from your data, and Vue updates the page when the data changes.
    CDNA website that hosts common files like Bootstrap and Vue so pages can load them. Offline, you use your own copy in vendor/.
    XSSAn attack where someone types code (e.g. <script>) into a form, hoping your page will run it. Showing text with textContent stops it.
    CSP (Content Security Policy)A line in the page's <head> that lists where the page may load scripts, images and data from. Everything else is blocked.
    CORSA browser rule: a page may call an API on another address only if the API says that's allowed.
    Git / commitGit keeps the history of your project. A commit is one saved point in that history, with a message saying what changed.

    L3The grid: container → row → col

    15 min
    1. .container centres the content with space at the sides (.container-fluid = full width).
    2. .row starts a line of columns. Only columns go directly inside a row.
    3. A row has 12 columns. col-4 takes 4 of 12 (a third), col-6 half, col-12 full width. Plain col shares the space equally; col-auto is as wide as its content.
    4. Your content goes inside the column. The space between columns comes from gutter classes on the row: g-0 … g-5.
    5. If the numbers add up to more than 12, the extra columns wrap to the next line. That's how cards flow into rows.

const API_BASE = 'http://localhost:4000';               // Node.js API (provided/api-server)

// ===== Login session =====
// After logging in, we keep the token and the user in sessionStorage.
// sessionStorage is emptied when the browser tab is closed.

// Saves the answer of a login or register request.
function saveSession(result) {
  const session = {
    token: result.token,
    expires_at: result.expires_at,
    user: result.user          // { id, name, email, role }
  };
  sessionStorage.setItem('session', JSON.stringify(session));
}

// Returns the saved session, or null when nobody is logged in or the token has expired.
function getSession() {
  const saved = sessionStorage.getItem('session');
  if (saved === null) {
    return null;
  }

  let session;
  try {
    session = JSON.parse(saved);
  } catch (error) {
    clearSession();            // the saved text was broken
    return null;
  }

  const expiresAt = new Date(session.expires_at);
  if (expiresAt < new Date()) {
    clearSession();            // the token is too old
    return null;
  }
  return session;
}

// Forgets the login.
function clearSession() {
  sessionStorage.removeItem('session');
}


// ===== Talking to the API =====

// Sends a request to the API and returns its answer as a JavaScript object.
//   path:     the part after /api, for example '/products?search=rice'
//   method:   'GET' (if left out), 'POST', 'PUT' or 'DELETE'
//   body:     data to send: an object (sent as JSON) or a FormData (for file uploads); or leave out
//   useToken: true to send the login token (needed for seller actions)
//
// When something goes wrong it throws an error with:
//   error.message  a text you can show to the user
//   error.status   the HTTP status, e.g. 401, 403, 422 (0 = the server could not be reached)
//   error.errors   the problems per form field, e.g. { email: ['The email has already been taken.'] }
async function apiFetch(path, method, body, useToken) {
  if (!method) {
    method = 'GET';
  }
  const options = {
    method: method,
    headers: {}
  };

  if (body instanceof FormData) {
    // A form with a file: send it as it is.
    // Don't set Content-Type: the browser adds "multipart/form-data" with the right boundary.
    options.body = body;
  } else if (body) {
    // Normal data: send it as JSON text
    options.headers['Content-Type'] = 'application/json';
    options.body = JSON.stringify(body);
  }

  if (useToken) {
    const session = getSession();
    if (session) {
      options.headers['Authorization'] = 'Bearer ' + session.token;
    }
  }

  let response;
  try {
    response = await fetch(API_URL + path, options);
  } catch (networkError) {
    // fetch() itself fails when the server is off or the address is wrong
    throw makeError(0, 'Cannot reach the server. Please check that the API is running and try again.', {});
  }

  // Turn the JSON answer into an object (a few answers have no body at all)
  let data = null;
  try {
    data = await response.json();
  } catch (error) {
    data = null;
  }

  // response.ok is false for status 400 and higher
  if (!response.ok) {
    if (response.status === 401 && useToken) {
      clearSession();            // the token is wrong or expired: log out here too
    }
    let message = 'Request failed (HTTP ' + response.status + ').';
    let errors = {};
    if (data && data.message) {
      message = data.message;
    }
    if (data && data.errors) {
      errors = data.errors;
    }
    throw makeError(response.status, message, errors);
  }

  return data;
}

// Creates an Error object that also carries the HTTP status and the field errors.
function makeError(status, message, errors) {
  const error = new Error(message);
  error.status = status;
  error.errors = errors;
  return error;
}

// Creates an element with a class and some text, for example:
//   makeElement('p', 'card-text', 'Hello')  makes  <p class="card-text">Hello</p>

function makeElement(tag, className, text) {

  if (className) {
    element.className = className;

  if (text !== undefined) {

}

// Marks a form field as wrong (red border, message under it) or as OK.
//   field:   the input element
//   message: the problem to show, or '' when the field is fine
// The message goes into the element whose id is the field's id + '-error',
// for example <div id="email-error" class="invalid-feedback">.
// Returns true when the field is wrong.
function markField(field, message) {
  if (message === '') {
    field.classList.remove('is-invalid');
    field.setAttribute('aria-invalid', 'false');
    return false;
  }
  field.classList.add('is-invalid');
  field.setAttribute('aria-invalid', 'true');
  const feedback = document.getElementById(field.id + '-error');
  if (feedback) {
    feedback.textContent = message;
  }
  return true;
}

// Shows the API's validation errors (status 422) under the matching form fields
// and puts the cursor in the first wrong field.
//   errors:  for example { email: ['The email has already been taken.'] }
//   renames: (optional) when an input's id is not the same as the API's field name,
//            for example { role: 'role-customer' }
function showServerErrors(errors, renames) {
  let firstField = null;
  for (const fieldName in errors) {
    let id = fieldName;
    if (renames && renames[fieldName]) {
      id = renames[fieldName];
    }
    const field = document.getElementById(id);
    if (field) {
      markField(field, errors[fieldName][0]);    // show the first message for that field
      if (firstField === null) {
        firstField = field;
      }
    }
  }
  if (firstField) {
    firstField.focus();
  }
}

// Builds the part after the "?" in the address, for example "page=1&sort=newest&search=rice".
// Empty choices are left out. encodeURIComponent() makes typed text safe inside an address.
function buildQuery() {
  let query = 'page=' + page + '&sort=' + sort;
  if (search !== '') {
    query = query + '&search=' + encodeURIComponent(search);
  }
  if (categoryId !== '') {
    query = query + '&category_id=' + categoryId;
  }
  if (minPrice !== '') {
    query = query + '&min_price=' + minPrice;
  }
  if (maxPrice !== '') {
    query = query + '&max_price=' + maxPrice;
  }
  return query;
}

// GET /api/products with the current choices, then shows the cards and the page buttons.

 *

 *   h('p', { class: 'fs-4' }, 'Hello')   makes   <p class="fs-4">Hello</p>
 * A Vue template would need 'unsafe-eval' in our Content Security Policy; h() doesn't.
 *

// Builds the Vue app. Runs once Vue has loaded.
function startVueSummary() {
  if (getSession() === null) {
    return;                         // not logged in: seller.js is sending the visitor to the login page
  }

  // The Vue functions we use
  const createApp = Vue.createApp;
  const ref = Vue.ref;              // a value that Vue watches: when it changes, the page updates
  const computed = Vue.computed;    // a value worked out from other values, updated automatically
  const onMounted = Vue.onMounted;  // runs code once the app is on the page
  const h = Vue.h;                  // builds HTML elements

  const app = createApp({
    setup: function () {
      const summary = ref(null);    // { products, units, stock_value, out_of_stock } from the API
      const errorText = ref('');

      // Loads the totals again from the API
      async function refresh() {
        try {
          const result = await apiFetch('/seller/summary', 'GET', null, true);
          summary.value = result.data;
          errorText.value = '';
        } catch (error) {
          errorText.value = 'Summary unavailable.';
        }
      }

      onMounted(function () {
        refresh();
        window.addEventListener('products-changed', refresh);   // seller.js sends this after each change
      });

      // A short sentence under the cards, worked out from the summary.
      // computed() makes Vue update it automatically whenever summary changes.
      const stockNote = computed(function () {
        if (summary.value === null) {
          return '';
        }
        if (summary.value.out_of_stock === 0) {
          return 'Everything is in stock.';
        }
        if (summary.value.out_of_stock === 1) {
          return '1 product is out of stock: add stock so customers can buy it.';
        }
        return summary.value.out_of_stock + ' products are out of stock: add stock so customers can buy them.';
      });

      // One Bootstrap card with a label and a big number.
      // col-6 col-md-3 = 2 cards per row on phones, 4 per row from 768px.
      function statCard(label, value) {

          h('div', { class: 'card h-100 text-center' }, [
            h('div', { class: 'card-body' }, [
              h('p', { class: 'text-body-secondary small mb-1' }, label),
              h('p', { class: 'fs-4 fw-bold mb-0' }, value)
            ])
          ])
        ]);
      }

      // What Vue shows. Vue runs this again whenever a value above changes.
      return function render() {
        if (errorText.value !== '') {
          return h('p', { class: 'text-body-secondary' }, errorText.value);
        }
        if (summary.value === null) {
          return h('p', { class: 'text-body-secondary' }, 'Loading summary…');   // no fake zeros while loading
        }
        return h('div', [
          h('div', { class: 'row g-3' }, [
            statCard('Products', String(summary.value.products)),
            statCard('Units in stock', String(summary.value.units)),
            statCard('Stock value', formatPeso(summary.value.stock_value)),
            statCard('Out of stock', String(summary.value.out_of_stock))
          ]),
          h('p', { class: 'small text-body-secondary mt-2 mb-0' }, stockNote.value)
        ]);
      };
    }
  });

// PUT /api/cart/:product_id { quantity }
async function changeQuantity(item, typedValue) {
  cartAlert.hidden = true;
  const quantity = Number(typedValue);

  // Check it first; a wrong value puts the old quantity back
  if (!Number.isInteger(quantity) || quantity < 1) {
    showAlert(cartAlert, 'warning', 'Enter a whole number, 1 or more. To take an item out, click Remove.');
    showCart(cart);
    return;
  }

  try {
    const result = await apiFetch('/cart/' + item.product_id, 'PUT', { quantity: quantity }, true);
    showCart(result.data);
  } catch (error) {
    if (error.status === 422 && error.errors.quantity) {
      showAlert(cartAlert, 'warning', error.errors.quantity[0]);    // e.g. "Only 18 left of Bangus (1 kg)."
      showCart(cart);
    } else {
      handleError(error);
    }
  }
}

// One order: a card with a header (number, date, status), a table of items, and the total.
function makeOrderCard(order) {
  // Header: "Order #2" on the left; date and status on the right
  const header = makeElement('div', 'card-header d-flex flex-wrap justify-content-between align-items-center gap-2');
  const title = makeElement('h2', 'h6 mb-0', 'Order #' + order.id);
  const info = makeElement('span', 'small text-body-secondary', formatDate(order.created_at) + ' ');
  info.append(makeElement('span', 'badge text-bg-success text-capitalize', order.status));
  header.append(title, info);

  // Items table
  const table = makeElement('table', 'table table-sm align-middle mb-0');
  const head = makeElement('thead');
  const headRow = makeElement('tr');
  const headings = ['Product', 'Price', 'Qty', 'Total'];
  for (const heading of headings) {
    const cell = makeElement('th', '', heading);
    cell.scope = 'col';
    if (heading !== 'Product') {
      cell.className = 'text-end';
    }
    headRow.append(cell);
  }
  head.append(headRow);

  const body = makeElement('tbody');
  for (const item of order.items) {
    const row = makeElement('tr');
    row.append(
      makeElement('td', '', item.name),
      makeElement('td', 'text-end text-nowrap', formatPeso(item.price)),
      makeElement('td', 'text-end', String(item.quantity)),
      makeElement('td', 'text-end text-nowrap', formatPeso(item.line_total))
    );
    body.append(row);
  }
  table.append(head, body);
  const tableWrap = makeElement('div', 'table-responsive');
  tableWrap.append(table);

  // Delivery details and the total
  const details = makeElement('div', 'card-body d-flex flex-column flex-md-row justify-content-between gap-2');

  details.append(delivery, total);

  const card = makeElement('article', 'card shadow-sm');
  card.append(header, tableWrap, details);
  return card;
}

// ---------- Passwords and tokens ----------
function hashPassword(password, salt = crypto.randomBytes(16).toString('hex')) {
  const hash = crypto.scryptSync(password, salt, 32).toString('hex');
  return { salt, hash };
}
function passwordMatches(user, password) {
  const { hash } = hashPassword(password, user.salt);
  return crypto.timingSafeEqual(Buffer.from(hash, 'hex'), Buffer.from(user.password_hash, 'hex'));
}
function issueToken(user) {
  const token = crypto.randomBytes(32).toString('hex');
  const expires_at = new Date(Date.now() + TOKEN_HOURS * 3600 * 1000).toISOString();
  db.tokens.push({ token, user_id: user.id, expires_at });
  save();
  return { token, token_type: 'Bearer', expires_at };
}
const publicUser = (u) => ({ id: u.id, name: u.name, email: u.email, role: u.role, created_at: u.created_at });

/** Returns the logged-in user from "Authorization: Bearer <token>", or null. */
function currentUser(req) {
  const match = /^Bearer\s+([a-f0-9]{64})$/i.exec(req.headers.authorization || '');
  if (!match) return null;
  const row = db.tokens.find((t) => t.token === match[1]);
  if (!row || new Date(row.expires_at) < new Date()) return null;
  return db.users.find((u) => u.id === row.user_id) || null;
}

// ---------- HTTP helpers ----------
function send(res, status, body) {
  const headers = {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization',
    'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
    'X-Content-Type-Options': 'nosniff'
  };
  if (body === undefined) { res.writeHead(status, headers); res.end(); return; }
  headers['Content-Type'] = 'application/json; charset=utf-8';
  res.writeHead(status, headers);
  res.end(JSON.stringify(body));
}
const fail = (res, status, message, errors) => send(res, status, errors ? { message, errors } : { message });

const httpError = (status, message) => Object.assign(new Error(message), { status });

/** Reads the whole request body as a Buffer; rejects bodies over MAX_BODY with 413. */
function readRaw(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    let size = 0;
    req.on('data', (chunk) => {
      size += chunk.length;
      if (size <= MAX_BODY) chunks.push(chunk);   // over the limit: keep reading but drop the data
    });
    // Answer only after the whole body has arrived, so the browser receives the 413 message
    req.on('end', () => (size > MAX_BODY ? reject(httpError(413, 'Request body too large (max 3 MB).')) : resolve(Buffer.concat(chunks))));
    req.on('error', reject);
  });
}

async function readJson(req) {
  const raw = (await readRaw(req)).toString('utf8');
  if (!raw) return {};
  try { return JSON.parse(raw); }
  catch { throw httpError(400, 'Request body must be valid JSON.'); }
}

/**
 * Parses multipart/form-data: text fields, plus one optional file in the field "image".
 * @returns {Promise<{fields: Record<string, string>, file: {filename: string, data: Buffer} | null}>}
 */
async function readMultipart(req) {
  const match = /boundary=(?:"([^"]+)"|([^;]+))/i.exec(req.headers['content-type'] || '');
  if (!match) throw httpError(400, 'Missing multipart boundary.');
  const boundary = Buffer.from('--' + (match[1] || match[2]).trim());
  const raw = await readRaw(req);
  const fields = {};
  let file = null;

  let pos = raw.indexOf(boundary);
  while (pos !== -1) {
    const start = pos + boundary.length;
    if (raw.slice(start, start + 2).toString() === '--') break;          // closing boundary
    const next = raw.indexOf(boundary, start);
    if (next === -1) break;
    const part = raw.slice(start + 2, next - 2);                         // drop the CRLF after and before the boundaries
    const headerEnd = part.indexOf('\r\n\r\n');
    if (headerEnd !== -1) {
      const headers = part.slice(0, headerEnd).toString('utf8');
      const data = part.slice(headerEnd + 4);
      const name = (/\bname="([^"]*)"/i.exec(headers) || [])[1];
      const filename = (/\bfilename="([^"]*)"/i.exec(headers) || [])[1];
      if (name === 'image' && filename) file = { filename, data };
      else if (name && filename === undefined) fields[name] = data.toString('utf8');
    }
    pos = next;
  }
  return { fields, file };
}

/**
 * Reads a product from JSON, or from multipart/form-data (text fields are converted to numbers).
 * @returns {Promise<{body: object, file: {filename: string, data: Buffer} | null}>}
 */
async function readProductBody(req) {
  if (!/^multipart\/form-data/i.test(req.headers['content-type'] || '')) return { body: await readJson(req), file: null };
  const { fields, file } = await readMultipart(req);
  const num = (v) => (v === undefined || v.trim() === '' ? NaN : Number(v));
  return {
    body: {
      name: fields.name, description: fields.description,
      price: num(fields.price), stock: num(fields.stock), category_id: num(fields.category_id),
      remove_image: fields.remove_image === 'true',
      _method: fields._method
    },
    file
  };
}

// ---------- Images ----------
/** JPG, PNG or WebP, detected from the file's first bytes (the name and declared type can lie). */
function imageExtension(buf) {
  if (buf.length > 3 && buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'jpg';
  if (buf.length > 8 && buf.slice(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) return 'png';
  if (buf.length > 12 && buf.slice(0, 4).toString() === 'RIFF' && buf.slice(8, 12).toString() === 'WEBP') return 'webp';
  return null;
}

/** Adds an "image" error when the uploaded file is not an allowed image or is too big. */
function validateImage(file, errors) {
  if (!file) return;
  if (!imageExtension(file.data)) errors.image = ['The image must be a JPG, PNG or WebP file.'];
  else if (file.data.length > MAX_IMAGE) errors.image = ['The image may not be larger than 2 MB.'];
}

/** Saves an uploaded image as <uuid>.<ext> and returns its public path. */
function saveImage(file) {
  const name = `${crypto.randomUUID()}.${imageExtension(file.data)}`;
  fs.writeFileSync(path.join(UPLOADS_DIR, name), file.data);
  return `/uploads/${name}`;
}

/** Deletes an uploaded image (only UUID files; the sample photos are never deleted). */
function deleteImage(imageUrl) {
  const name = imageUrl ? path.basename(imageUrl) : '';
  if (UUID_IMAGE.test(name) && fs.existsSync(path.join(UPLOADS_DIR, name))) fs.unlinkSync(path.join(UPLOADS_DIR, name));
}

// ---------- Validation ----------
const isEmail = (v) => typeof v === 'string' && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v);
function validateRegister(b) {
  const errors = {};
  if (typeof b.name !== 'string' || b.name.trim().length < 2 || b.name.trim().length > 60) errors.name = ['The name must be 2 to 60 characters.'];
  if (!isEmail(b.email)) errors.email = ['The email must be a valid email address.'];
  else if (db.users.some((u) => u.email.toLowerCase() === b.email.toLowerCase())) errors.email = ['The email has already been taken.'];
  if (typeof b.password !== 'string' || b.password.length < 8 || !/[A-Za-z]/.test(b.password) || !/\d/.test(b.password)) {
    errors.password = ['The password must be at least 8 characters and contain a letter and a number.'];
  }
  if (!['customer', 'seller'].includes(b.role)) errors.role = ['The role must be customer or seller.'];
  return errors;
}
function validateProduct(b) {
  const errors = {};
  if (typeof b.name !== 'string' || b.name.trim().length < 3 || b.name.trim().length > 80) errors.name = ['The name must be 3 to 80 characters.'];
  if (typeof b.price !== 'number' || !Number.isFinite(b.price) || b.price <= 0) errors.price = ['The price must be a number greater than 0.'];
  if (!Number.isInteger(b.stock) || b.stock < 0) errors.stock = ['The stock must be a whole number, 0 or more.'];
  if (!db.categories.some((c) => c.id === b.category_id)) errors.category_id = ['The selected category is invalid.'];
  if (b.description !== undefined && (typeof b.description !== 'string' || b.description.length > 500)) errors.description = ['The description may not be longer than 500 characters.'];
  if (b.image_url !== undefined && b.image_url !== null && (typeof b.image_url !== 'string' || !/^\/uploads\/[\w.-]+$/.test(b.image_url))) {
    errors.image_url = ['The image_url must be a path like /uploads/file.webp.'];
  }
  return errors;
}

const productOut = (p) => {
  const seller = db.users.find((u) => u.id === p.seller_id);
  return { ...p, seller_name: seller ? seller.name : null };
};

// ---------- Routes ----------
async function handle(req, res) {
  const url = new URL(req.url, `http://localhost:${PORT}`);
  const parts = url.pathname.replace(/\/+$/, '').split('/').filter(Boolean);   // e.g. ['api','products','3']
  const method = req.method;

  if (method === 'OPTIONS') return send(res, 204);

  // Static product images: GET /uploads/<file>
  if (method === 'GET' && parts[0] === 'uploads' && parts.length === 2) {
    const file = path.join(UPLOADS_DIR, path.basename(parts[1]));
    if (!fs.existsSync(file)) return fail(res, 404, 'Image not found.');
    const types = { '.jpg': 'image/jpeg', '.png': 'image/png', '.webp': 'image/webp' };
    const type = types[path.extname(file).toLowerCase()];
    if (!type) return fail(res, 404, 'Image not found.');
    res.writeHead(200, { 'Content-Type': type, 'Access-Control-Allow-Origin': '*', 'Cache-Control': 'max-age=3600', 'X-Content-Type-Options': 'nosniff' });
    return fs.createReadStream(file).pipe(res);
  }

  if (parts[0] !== 'api') return fail(res, 404, 'Not found. All endpoints start with /api.');
  const route = parts.slice(1);

  // GET /api/health
  if (method === 'GET' && route[0] === 'health' && route.length === 1) return send(res, 200, { status: 'ok', time: now() });

  // ----- Auth -----
  if (route[0] === 'auth') {
    if (method === 'POST' && route[1] === 'register') {
      const b = await readJson(req);
      const errors = validateRegister(b);
      if (Object.keys(errors).length) return fail(res, 422, 'The given data was invalid.', errors);
      const { salt, hash } = hashPassword(b.password);
      const user = { id: nextId(db.users), name: b.name.trim(), email: b.email.trim().toLowerCase(), role: b.role, salt, password_hash: hash, created_at: now() };
      db.users.push(user);
      save();
      return send(res, 201, { message: 'Registered successfully.', ...issueToken(user), user: publicUser(user) });
    }
    if (method === 'POST' && route[1] === 'login') {
      const b = await readJson(req);
      if (!isEmail(b.email) || typeof b.password !== 'string' || !b.password) {
        return fail(res, 422, 'The given data was invalid.', { email: ['Email and password are required.'] });
      }
      const user = db.users.find((u) => u.email === b.email.trim().toLowerCase());
      if (!user || !passwordMatches(user, b.password)) return fail(res, 401, 'Invalid email or password.');
      return send(res, 200, { message: 'Logged in successfully.', ...issueToken(user), user: publicUser(user) });
    }
    if (method === 'POST' && route[1] === 'logout') {
      const user = currentUser(req);
      if (!user) return fail(res, 401, 'Unauthenticated.');
      const token = req.headers.authorization.split(' ')[1];
      db.tokens = db.tokens.filter((t) => t.token !== token);
      save();
      return send(res, 200, { message: 'Logged out.' });
    }
    if (method === 'GET' && route[1] === 'me') {
      const user = currentUser(req);
      if (!user) return fail(res, 401, 'Unauthenticated.');
      return send(res, 200, { data: publicUser(user) });
    }
  }

  // ----- Categories -----
  if (route[0] === 'categories' && method === 'GET') {
    if (route.length === 1) return send(res, 200, { data: db.categories });
    // GET /api/categories/:id/products -> same as /api/products?category_id=:id
    if (route.length === 3 && route[2] === 'products') {
      const id = Number(route[1]);
      if (!db.categories.some((c) => c.id === id)) return fail(res, 404, 'Category not found.');
      url.searchParams.set('category_id', String(id));
      return listProducts(res, url);
    }
  }

  // ----- Products (public) -----
  if (route[0] === 'products' && method === 'GET') {
    if (route.length === 1) return listProducts(res, url);
    if (route.length === 2) {
      const p = db.products.find((x) => x.id === Number(route[1]));
      return p ? send(res, 200, { data: productOut(p) }) : fail(res, 404, 'Product not found.');
    }
  }

  // ----- Products (seller only) -----
  const needsSeller = (route[0] === 'products' && ['POST', 'PUT', 'DELETE'].includes(method)) ||
                      (route[0] === 'seller' && (route[1] === 'products' || route[1] === 'summary'));
  if (needsSeller) {
    const user = currentUser(req);
    if (!user) return fail(res, 401, 'Unauthenticated.');
    if (user.role !== 'seller') return fail(res, 403, 'This action is for sellers only.');

    // GET /api/seller/products: the seller's own products, with the same search/filters/pages as /api/products
    if (method === 'GET' && route[0] === 'seller' && route[1] === 'products' && route.length === 2) {
      return listProducts(res, url, user.id);
    }
    // GET /api/seller/summary: totals over ALL the seller's products
    if (method === 'GET' && route[0] === 'seller' && route[1] === 'summary' && route.length === 2) {
      const mine = db.products.filter((p) => p.seller_id === user.id);
      return send(res, 200, { data: {
        products: mine.length,
        units: mine.reduce((sum, p) => sum + p.stock, 0),
        stock_value: Math.round(mine.reduce((sum, p) => sum + p.price * p.stock, 0) * 100) / 100,
        out_of_stock: mine.filter((p) => p.stock === 0).length
      } });
    }
    if (method === 'POST' && route.length === 1) {
      const { body: b, file } = await readProductBody(req);
      const errors = validateProduct(b);
      validateImage(file, errors);
      if (Object.keys(errors).length) return fail(res, 422, 'The given data was invalid.', errors);
      const p = {
        id: nextId(db.products), name: b.name.trim(), description: (b.description || '').trim(), price: Math.round(b.price * 100) / 100,
        stock: b.stock, category_id: b.category_id, image_url: file ? saveImage(file) : (b.image_url || null),
        seller_id: user.id, created_at: now(), updated_at: now()
      };
      db.products.push(p);
      save();
      return send(res, 201, { message: 'Product created.', data: productOut(p) });
    }
    // PUT /api/products/:id, or POST /api/products/:id with _method=PUT (forms with a photo, as PHP needs)
    if ((method === 'PUT' || method === 'DELETE' || method === 'POST') && route.length === 2) {
      const p = db.products.find((x) => x.id === Number(route[1]));
      if (!p) return fail(res, 404, 'Product not found.');
      if (p.seller_id !== user.id) return fail(res, 403, 'You can only change your own products.');
      if (method === 'DELETE') {
        deleteImage(p.image_url);
        db.products = db.products.filter((x) => x.id !== p.id);
        db.cart_items = db.cart_items.filter((c) => c.product_id !== p.id);   // also take it out of every cart
        save();
        return send(res, 200, { message: 'Product deleted.' });
      }
      const { body: b, file } = await readProductBody(req);
      if (method === 'POST' && b._method !== 'PUT') return fail(res, 404, `No endpoint for POST ${url.pathname}.`);
      const errors = validateProduct(b);
      validateImage(file, errors);
      if (Object.keys(errors).length) return fail(res, 422, 'The given data was invalid.', errors);
      // New photo: replace (and delete the old file). remove_image: no photo. Otherwise keep the current one.
      let imageUrl = p.image_url;
      if (file) { deleteImage(p.image_url); imageUrl = saveImage(file); }
      else if (b.remove_image) { deleteImage(p.image_url); imageUrl = null; }
      else if (b.image_url !== undefined) imageUrl = b.image_url;
      Object.assign(p, {
        name: b.name.trim(), description: (b.description || '').trim(), price: Math.round(b.price * 100) / 100,
        stock: b.stock, category_id: b.category_id, image_url: imageUrl, updated_at: now()
      });
      save();
      return send(res, 200, { message: 'Product updated.', data: productOut(p) });
    }
  }

  // ----- Cart and orders (customers only) -----
  if (route[0] === 'cart' || route[0] === 'orders') {
    const user = currentUser(req);
    if (!user) return fail(res, 401, 'Unauthenticated.');
    if (user.role !== 'customer') return fail(res, 403, 'This action is for customers only.');

    if (route[0] === 'cart') {
      // GET /api/cart
      if (method === 'GET' && route.length === 1) return send(res, 200, { data: cartOut(user.id) });

      // POST /api/cart { product_id, quantity }: adds to the line if the product is already in the cart
      if (method === 'POST' && route.length === 1) {
        const b = await readJson(req);
        const quantity = b.quantity === undefined ? 1 : b.quantity;
        const product = db.products.find((p) => p.id === b.product_id);
        if (!product) return fail(res, 422, 'The given data was invalid.', { product_id: ['The selected product is invalid.'] });
        if (!Number.isInteger(quantity) || quantity < 1) return fail(res, 422, 'The given data was invalid.', { quantity: ['The quantity must be a whole number, 1 or more.'] });
        const line = db.cart_items.find((c) => c.user_id === user.id && c.product_id === product.id);
        const wanted = (line ? line.quantity : 0) + quantity;
        const stockError = stockProblem(product, wanted);
        if (stockError) return fail(res, 422, 'The given data was invalid.', { quantity: [stockError] });
        if (line) line.quantity = wanted;
        else db.cart_items.push({ user_id: user.id, product_id: product.id, quantity: wanted });
        save();
        return send(res, 200, { message: 'Added to cart.', data: cartOut(user.id) });
      }

      // PUT /api/cart/:product_id { quantity } and DELETE /api/cart/:product_id
      if ((method === 'PUT' || method === 'DELETE') && route.length === 2) {
        const productId = Number(route[1]);
        const line = db.cart_items.find((c) => c.user_id === user.id && c.product_id === productId);
        if (!line) return fail(res, 404, 'This product is not in your cart.');
        if (method === 'DELETE') {
          db.cart_items = db.cart_items.filter((c) => c !== line);
          save();
          return send(res, 200, { message: 'Removed from cart.', data: cartOut(user.id) });
        }
        const b = await readJson(req);
        if (!Number.isInteger(b.quantity) || b.quantity < 1) return fail(res, 422, 'The given data was invalid.', { quantity: ['The quantity must be a whole number, 1 or more.'] });
        const stockError = stockProblem(db.products.find((p) => p.id === productId), b.quantity);
        if (stockError) return fail(res, 422, 'The given data was invalid.', { quantity: [stockError] });
        line.quantity = b.quantity;
        save();
        return send(res, 200, { message: 'Cart updated.', data: cartOut(user.id) });
      }
    }

    if (route[0] === 'orders') {
      // GET /api/orders: the customer's orders, newest first
      if (method === 'GET' && route.length === 1) {
        const mine = db.orders.filter((o) => o.user_id === user.id).sort((a, b) => b.id - a.id).map(orderOut);
        return send(res, 200, { data: mine });
      }
      // GET /api/orders/:id
      if (method === 'GET' && route.length === 2) {
        const order = db.orders.find((o) => o.id === Number(route[1]) && o.user_id === user.id);
        return order ? send(res, 200, { data: orderOut(order) }) : fail(res, 404, 'Order not found.');
      }
      // POST /api/orders { address, notes }: checkout. Prices and totals come from the server, never the browser.
      if (method === 'POST' && route.length === 1) {
        const b = await readJson(req);
        const errors = {};
        const address = typeof b.address === 'string' ? b.address.trim() : '';
        const notes = typeof b.notes === 'string' ? b.notes.trim() : '';
        if (address.length < 5 || address.length > 200) errors.address = ['The delivery address must be 5 to 200 characters.'];
        if (notes.length > 200) errors.notes = ['The notes may not be longer than 200 characters.'];
        const lines = db.cart_items.filter((c) => c.user_id === user.id);
        if (!lines.length) errors.cart = ['Your cart is empty.'];
        for (const line of lines) {
          const product = db.products.find((p) => p.id === line.product_id);
          const stockError = stockProblem(product, line.quantity);
          if (stockError) { errors.cart = [stockError]; break; }
        }
        if (Object.keys(errors).length) return fail(res, 422, 'The given data was invalid.', errors);

        const items = lines.map((line) => {
          const product = db.products.find((p) => p.id === line.product_id);
          product.stock -= line.quantity;                                   // the stock goes down
          return { product_id: product.id, name: product.name, price: product.price, quantity: line.quantity,
                   line_total: Math.round(product.price * line.quantity * 100) / 100 };
        });
        const order = {
          id: nextId(db.orders), user_id: user.id, address, notes, status: 'placed',
          total: Math.round(items.reduce((sum, i) => sum + i.line_total, 0) * 100) / 100,
          created_at: now(), items
        };
        db.orders.push(order);
        db.cart_items = db.cart_items.filter((c) => c.user_id !== user.id);   // empty the cart
        save();
        return send(res, 201, { message: 'Order placed.', data: orderOut(order) });
      }
    }
  }

  return fail(res, 404, `No endpoint for ${method} ${url.pathname}.`);
}

  if (!product) return 'A product in your cart is no longer available.';
  if (product.stock === 0) return `${product.name} is out of stock.`;
  if (quantity > product.stock) return `Only ${product.stock} left of ${product.name}.`;
  return '';
}

  const q = url.searchParams;
  const stock = q.get('stock') || '';   // '' (all), 'in' (in stock) or 'out' (out of stock)
  const search = (q.get('search') || '').trim().toLowerCase();
  const categoryId = q.get('category_id') ? Number(q.get('category_id')) : null;
  const min = q.get('min_price') !== null && q.get('min_price') !== '' ? Number(q.get('min_price')) : null;
  const max = q.get('max_price') !== null && q.get('max_price') !== '' ? Number(q.get('max_price')) : null;
  const sort = q.get('sort') || 'newest';
  const page = Math.max(1, parseInt(q.get('page'), 10) || 1);
  const limit = Math.min(50, Math.max(1, parseInt(q.get('limit'), 10) || 8));

  if ([min, max].some((n) => n !== null && !Number.isFinite(n))) {
    return fail(res, 422, 'The given data was invalid.', { price: ['min_price and max_price must be numbers.'] });
  }
  if (!['newest', 'price_asc', 'price_desc', 'name'].includes(sort)) {
    return fail(res, 422, 'The given data was invalid.', { sort: ['sort must be newest, price_asc, price_desc or name.'] });
  }
  if (!['', 'in', 'out'].includes(stock)) {
    return fail(res, 422, 'The given data was invalid.', { stock: ['stock must be in or out.'] });
  }

  let list = db.products.filter((p) =>
    (!search || p.name.toLowerCase().includes(search) || (p.description || '').toLowerCase().includes(search)) &&
    (categoryId === null || p.category_id === categoryId) &&
    (min === null || p.price >= min) &&
    (max === null || p.price <= max) &&
    (sellerId === null || p.seller_id === sellerId) &&
    (stock === '' || (stock === 'in' ? p.stock > 0 : p.stock === 0)));
  const sorters = {
    newest: (a, b) => b.created_at.localeCompare(a.created_at) || b.id - a.id,
    price_asc: (a, b) => a.price - b.price || a.id - b.id,
    price_desc: (a, b) => b.price - a.price || a.id - b.id,
    name: (a, b) => a.name.localeCompare(b.name)
  };
  list = list.sort(sorters[sort]);
  const total = list.length;
  const data = list.slice((page - 1) * limit, page * limit).map(productOut);
  return send(res, 200, { data, meta: { page, limit, total, total_pages: Math.max(1, Math.ceil(total / limit)) } });
}

function add_to_cart() {
    $user = require_customer();
    $data = read_json();
    $product = select_one('SELECT * FROM products WHERE id = ?', 'i', [(int) ($data['product_id'] ?? 0)]);
    $quantity = $data['quantity'] ?? 1;
    if (!$product) {
        fail(422, 'The given data was invalid.', ['product_id' => ['The selected product is invalid.']]);
    }
    if (!is_positive_whole_number($quantity)) {
        fail(422, 'The given data was invalid.', ['quantity' => ['The quantity must be a whole number, 1 or more.']]);
    }

    $line = select_one('SELECT quantity FROM cart_items WHERE user_id = ? AND product_id = ?', 'ii', [$user['id'], $product['id']]);
    $already = $line ? (int) $line['quantity'] : 0;
    $wanted = $already + (int) $quantity;
    $problem = stock_problem($product, $wanted);
    if ($problem) {
        fail(422, 'The given data was invalid.', ['quantity' => [$problem]]);
    }

    if ($line) {
        run_query('UPDATE cart_items SET quantity = ? WHERE user_id = ? AND product_id = ?', 'iii', [$wanted, $user['id'], $product['id']]);
    } else {
        run_query('INSERT INTO cart_items (user_id, product_id, quantity) VALUES (?, ?, ?)', 'iii', [$user['id'], $product['id'], $wanted]);

}

// POST /api/orders  { address, notes }: checkout.
// The prices and the total come from the database, never from the browser.
function place_order() {
    global $conn;
    $user = require_customer();
    $data = read_json();
    $address = trim($data['address'] ?? '');
    $notes = trim($data['notes'] ?? '');

    $errors = [];
    if (mb_strlen($address) < 5 || mb_strlen($address) > 200) {
        $errors['address'] = ['The delivery address must be 5 to 200 characters.'];
    }
    if (mb_strlen($notes) > 200) {
        $errors['notes'] = ['The notes may not be longer than 200 characters.'];
    }
    $lines = select_all(
        'SELECT c.product_id, c.quantity, p.name, p.price, p.stock
         FROM cart_items c JOIN products p ON p.id = c.product_id
         WHERE c.user_id = ? ORDER BY c.id',
        'i', [$user['id']]
    );
    if (!$lines) {
        $errors['cart'] = ['Your cart is empty.'];
    }
    foreach ($lines as $line) {
        $problem = stock_problem($line, (int) $line['quantity']);
        if ($problem) {
            $errors['cart'] = [$problem];
            break;
        }
    }
    if ($errors) {
        fail(422, 'The given data was invalid.', $errors);
    }

    $total = 0;
    foreach ($lines as $line) {
        $total = $total + round((float) $line['price'] * (int) $line['quantity'], 2);
    }

    // A transaction: the order, its items, the stock changes and emptying the cart
    // are saved together, or (if anything fails) none of them is saved.
    mysqli_begin_transaction($conn);
    run_query('INSERT INTO orders (user_id, address, notes, total) VALUES (?, ?, ?, ?)',
        'issd', [$user['id'], $address, $notes, round($total, 2)]);
    $order_id = mysqli_insert_id($conn);

    foreach ($lines as $line) {
        // Lower the stock only if there is still enough (another customer may have just bought some)
        $stmt = run_query('UPDATE products SET stock = stock - ? WHERE id = ? AND stock >= ?',
            'iii', [$line['quantity'], $line['product_id'], $line['quantity']]);
        if (mysqli_stmt_affected_rows($stmt) !== 1) {
            mysqli_rollback($conn);
            fail(422, 'The given data was invalid.', ['cart' => ['Only ' . $line['stock'] . ' left of ' . $line['name'] . '.']]);
        }
        run_query('INSERT INTO order_items (order_id, product_id, name, price, quantity) VALUES (?, ?, ?, ?, ?)',
            'iisdi', [$order_id, $line['product_id'], $line['name'], $line['price'], $line['quantity']]);
    }
    run_query('DELETE FROM cart_items WHERE user_id = ?', 'i', [$user['id']]);
    mysqli_commit($conn);

    $order = select_one('SELECT * FROM orders WHERE id = ?', 'i', [$order_id]);
    send_json(201, ['message' => 'Order placed.', 'data' => order_out($order)]);
}

// Reads a JSON request body into an array.
function read_json() {
    $raw = file_get_contents('php://input');
    if ($raw === '') {
        return [];
    }
    $data = json_decode($raw, true);
    if (!is_array($data)) {

    return $data;

// Runs a query safely. The values go in separately from the SQL (a prepared statement),
// so text typed by users can never change the SQL (no SQL injection).
// $types has one letter per value: s = text, i = whole number, d = decimal number.
// Example: run_query('DELETE FROM products WHERE id = ?', 'i', [5]);
function run_query($sql, $types = '', $values = []) {
    global $conn;
    $stmt = mysqli_prepare($conn, $sql);
    if ($types !== '') {
        mysqli_stmt_bind_param($stmt, $types, ...$values);
    }
    mysqli_stmt_execute($stmt);
    return $stmt;
}

// Returns all rows of a SELECT as arrays, e.g. [['id' => 1, 'name' => 'Vegetables'], ...].
function select_all($sql, $types = '', $values = []) {
    $result = mysqli_stmt_get_result(run_query($sql, $types, $values));
    return mysqli_fetch_all($result, MYSQLI_ASSOC);
}

// Returns the first row of a SELECT, or null when there is none.
function select_one($sql, $types = '', $values = []) {
    $rows = select_all($sql, $types, $values);
    return $rows ? $rows[0] : null;
}

// The token from the "Authorization: Bearer <token>" header, or null.
function bearer_token() {
    $header = $_SERVER['HTTP_AUTHORIZATION'] ?? ($_SERVER['REDIRECT_HTTP_AUTHORIZATION'] ?? '');
    if ($header === '' && function_exists('getallheaders')) {
        foreach (getallheaders() as $name => $value) {
            if (strtolower($name) === 'authorization') {
                $header = $value;
            }
        }
    }
    return preg_match('/^Bearer\s+([a-f0-9]{64})$/i', $header, $match) ? $match[1] : null;
}

// Finds the user who owns the token (if it hasn't expired), or returns null.
function current_user() {
    $token = bearer_token();
    if (!$token) {
        return null;
    }
    return select_one(
        'SELECT users.* FROM tokens JOIN users ON users.id = tokens.user_id
         WHERE tokens.token = ? AND tokens.expires_at > ?',
        'ss', [$token, date('Y-m-d H:i:s')]
    );
}

// Creates a random login token for a user and saves it.
function issue_token($user) {
    global $TOKEN_HOURS;
    $token = bin2hex(random_bytes(32));                       // 64 random hexadecimal characters
    $expires = time() + $TOKEN_HOURS * 3600;

}

// A random UUID (version 4), e.g. "3f2c9a1e-7b4d-4c1a-9e2f-5a6b7c8d9e0f".
function uuid_v4() {
    $bytes = random_bytes(16);
    $bytes[6] = chr((ord($bytes[6]) & 0x0f) | 0x40);   // version 4
    $bytes[8] = chr((ord($bytes[8]) & 0x3f) | 0x80);   // variant
    $hex = bin2hex($bytes);
    return substr($hex, 0, 8) . '-' . substr($hex, 8, 4) . '-' . substr($hex, 12, 4) . '-'
         . substr($hex, 16, 4) . '-' . substr($hex, 20, 12);
}

// Checks the uploaded "image" file. Returns an error message, or '' when it is fine (or not sent).
function check_image() {
    global $MAX_IMAGE_BYTES;
    if (!isset($_FILES['image']) || $_FILES['image']['error'] === UPLOAD_ERR_NO_FILE) {
        return '';
    }
    $file = $_FILES['image'];
    if ($file['error'] === UPLOAD_ERR_INI_SIZE || $file['error'] === UPLOAD_ERR_FORM_SIZE || $file['size'] > $MAX_IMAGE_BYTES) {
        return 'The image may not be larger than 2 MB.';
    }
    if ($file['error'] !== UPLOAD_ERR_OK) {
        return 'The image could not be uploaded.';
    }
    // getimagesize() reads the file itself, so a text file renamed to .jpg is refused
    $info = @getimagesize($file['tmp_name']);
    if (!$info || !in_array($info['mime'], ['image/jpeg', 'image/png', 'image/webp'])) {
        return 'The image must be a JPG, PNG or WebP file.';
    }
    return '';
}

// Saves the uploaded "image" as uploads/<uuid>.<ext> and returns its path, or null when none was sent.
function save_image() {
    if (!isset($_FILES['image']) || $_FILES['image']['error'] !== UPLOAD_ERR_OK) {
        return null;
    }
    $extensions = ['image/jpeg' => 'jpg', 'image/png' => 'png', 'image/webp' => 'webp'];
    $mime = getimagesize($_FILES['image']['tmp_name'])['mime'];
    $name = uuid_v4() . '.' . $extensions[$mime];
    move_uploaded_file($_FILES['image']['tmp_name'], __DIR__ . '/uploads/' . $name);
    return '/uploads/' . $name;
}

// Deletes an uploaded photo. Only UUID files are deleted, never the sample photos.
function delete_image($image_url) {
    if ($image_url && preg_match('#^/uploads/[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\.(jpg|png|webp)$#', $image_url)) {
        $path = __DIR__ . $image_url;
        if (is_file($path)) {
            unlink($path);
        }
    }
}
